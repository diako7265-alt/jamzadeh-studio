        function playBeep(freq, duration, type = 'sine') {
            if (!audioEnabled) return;
            try {
                if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                if (audioCtx.state === 'suspended') audioCtx.resume();
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = type;
                osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
                gain.gain.setValueAtTime(0.04, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + duration);
                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start();
                osc.stop(audioCtx.currentTime + duration);

        }

        function acesFilmicToneMapping(x) {
            const a = 2.51;
            const b = 0.03;
            const c = 2.43;
            const d = 0.59;
            const e = 0.14;
            return Math.min(255, Math.max(0, ((x * (a * x + b)) / (x * (c * x + d) + e)) * 255));
        }

        function computeVRayRenderPipeline(sourceImg = null) {
            const img = sourceImg || baseImage;
            const mainCanvas = document.getElementById('vrayCanvas');
            const w = img.naturalWidth || img.width || 1200;
            const h = img.naturalHeight || img.height || 800;

            mainCanvas.width = w;
            mainCanvas.height = h;

            Object.values(channelsCanvas).forEach(c => {
                c.width = w;
                c.height = h;
            });

            // 1. Raw Base Image Pass
            const wireCtx = channelsCanvas.wireframe.getContext('2d');
            wireCtx.drawImage(img, 0, 0, w, h);
            
            let d;
            try {
                const imgData = wireCtx.getImageData(0, 0, w, h);
                d = imgData.data;
            } catch(e) {
                const rgbCtx = channelsCanvas.rgb.getContext('2d');
                rgbCtx.drawImage(img, 0, 0, w, h);
                renderActiveChannelToScreen();
                return;
            }

            // 2. Prepare Secondary Render Channels
            const rgbCtx = channelsCanvas.rgb.getContext('2d');
            const aoCtx = channelsCanvas.ao.getContext('2d');
            const normCtx = channelsCanvas.normals.getContext('2d');

            const aoData = aoCtx.createImageData(w, h);
            const normData = normCtx.createImageData(w, h);
            const beautyData = rgbCtx.createImageData(w, h);

            const useTravertine = document.getElementById('matTravertine').checked;
            const useConcrete = document.getElementById('matConcrete').checked;
            const useGlass = document.getElementById('matCurtainWall').checked;

            for (let y = 1; y < h - 1; y++) {
                for (let x = 1; x < w - 1; x++) {
                    const idx = (y * w + x) * 4;

                    const idxLeft = (y * w + (x - 1)) * 4;
                    const idxRight = (y * w + (x + 1)) * 4;
                    const idxUp = ((y - 1) * w + x) * 4;
                    const idxDown = ((y + 1) * w + x) * 4;

                    const lumL = 0.299 * d[idxLeft] + 0.587 * d[idxLeft + 1] + 0.114 * d[idxLeft + 2];
                    const lumR = 0.299 * d[idxRight] + 0.587 * d[idxRight + 1] + 0.114 * d[idxRight + 2];
                    const lumU = 0.299 * d[idxUp] + 0.587 * d[idxUp + 1] + 0.114 * d[idxUp + 2];
                    const lumD = 0.299 * d[idxDown] + 0.587 * d[idxDown + 1] + 0.114 * d[idxDown + 2];

                    const dx = (lumR - lumL);
                    const dy = (lumD - lumU);

                    normData.data[idx] = Math.min(255, Math.max(0, 128 + dx * 2.2));
                    normData.data[idx + 1] = Math.min(255, Math.max(0, 128 + dy * 2.2));
                    normData.data[idx + 2] = 255;
                    normData.data[idx + 3] = 255;

                    const edgeStrength = Math.abs(dx) + Math.abs(dy);
                    const aoVal = Math.max(25, 255 - edgeStrength * 2.4);
                    aoData.data[idx] = aoVal;
                    aoData.data[idx + 1] = aoVal;
                    aoData.data[idx + 2] = aoVal;
                    aoData.data[idx + 3] = 255;

                    let r = d[idx] / 255.0;
                    let g = d[idx + 1] / 255.0;
                    let b = d[idx + 2] / 255.0;

                    const pseudoRandom = Math.sin(x * 12.9898 + y * 78.233) * 43758.5453;
                    const microGrain = (pseudoRandom - Math.floor(pseudoRandom)) * 0.08 - 0.04;

                    if (useTravertine || useConcrete) {
                        r += microGrain;
                        g += microGrain;
                        b += microGrain;
                    }

                    const aoFactor = aoVal / 255.0;
                    r *= (0.35 + 0.65 * aoFactor);
                    g *= (0.35 + 0.65 * aoFactor);
                    b *= (0.35 + 0.65 * aoFactor);

                    if (currentLighting === 'golden') {
                        r = r * 1.35 + 0.08;
                        g = g * 1.08 + 0.03;
                        b = b * 0.78;
                    } else if (currentLighting === 'night') {
                        const isWarmLight = (r > 0.55 && g > 0.45);
                        if (isWarmLight) {
                            r = r * 1.45 + 0.12;
                            g = g * 1.25 + 0.06;
                            b = b * 0.85;
                        } else {
                            r = r * 0.42;
                            g = g * 0.52 + 0.02;
                            b = b * 0.85 + 0.08;
                        }
                    } else if (currentLighting === 'rain') {
                        const isGround = y > (h * 0.6);
                        if (isGround && useGlass) {
                            r = r * 1.25 + 0.05;
                            g = g * 1.30 + 0.06;
                            b = b * 1.40 + 0.10;
                        } else {
                            r = r * 0.82;
                            g = g * 0.94;
                            b = b * 1.08;
                        }
                    } else {
                        r = r * 1.15;
                        g = g * 1.12;
                        b = b * 1.10;
                    }

                    beautyData.data[idx] = acesFilmicToneMapping(r);
                    beautyData.data[idx + 1] = acesFilmicToneMapping(g);
                    beautyData.data[idx + 2] = acesFilmicToneMapping(b);
                    beautyData.data[idx + 3] = 255;
                }
            }

            aoCtx.putImageData(aoData, 0, 0);
            normCtx.putImageData(normData, 0, 0);
            rgbCtx.putImageData(beautyData, 0, 0);

            rgbCtx.globalCompositeOperation = 'screen';
            const bloomGrad = rgbCtx.createRadialGradient(w * 0.72, h * 0.28, 10, w * 0.72, h * 0.28, w * 0.65);
            if (currentLighting === 'golden') {
                bloomGrad.addColorStop(0, 'rgba(255, 220, 160, 0.45)');
            } else if (currentLighting === 'night') {
                bloomGrad.addColorStop(0, 'rgba(255, 200, 110, 0.40)');
            } else {
                bloomGrad.addColorStop(0, 'rgba(230, 245, 255, 0.35)');
            }
            bloomGrad.addColorStop(1, 'transparent');
            rgbCtx.fillStyle = bloomGrad;
            rgbCtx.fillRect(0, 0, w, h);
            rgbCtx.globalCompositeOperation = 'source-over';

            renderActiveChannelToScreen();
        }

        function initSplitter() {
            const container = document.getElementById('viewportDisplayWrapper');
            const divider = document.getElementById('curtainDivider');
            const afterWrapper = document.getElementById('afterClippedBox');
            let isDragging = false;

            function updateSplit(clientX) {
                const rect = container.getBoundingClientRect();
                let x = clientX - rect.left;
                if (x < 0) x = 0;
                if (x > rect.width) x = rect.width;
                
                const percent = (x / rect.width) * 100;
                divider.style.left = `${percent}%`;
                afterWrapper.style.clipPath = `polygon(${percent}% 0, 100% 0, 100% 100%, ${percent}% 100%)`;
            }

            container.addEventListener('mousedown', (e) => {
                isDragging = true;
                updateSplit(e.clientX);
            });
            window.addEventListener('mousemove', (e) => {
                if (isDragging) updateSplit(e.clientX);
            });
            window.addEventListener('mouseup', () => isDragging = false);

            container.addEventListener('touchstart', (e) => {
                isDragging = true;
                updateSplit(e.touches[0].clientX);
            });
            window.addEventListener('touchmove', (e) => {
                if (isDragging) updateSplit(e.touches[0].clientX);
            });
            window.addEventListener('touchend', () => isDragging = false);
        }
